  const DELAY_THRESHOLDS = {
    INITIAL_REVIEW: 90,
  // 初审：90天

  // 普通稿：30天
      urgent: 15,
  // 急稿：15天
      very_urgent: 7
  // 重要急稿：7天
    }
  };

  判断逻辑

  IF 当前节点状态 ==
  IN_PROGRESS
     AND (当前日期 -
  开始时间) > 阈值
  THEN
     项目状态 = DELAYED
     显示延期提醒
  END
